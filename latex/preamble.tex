% =============================================================================
%  preamble.tex -- packages, theorem environments, layout
%  Loaded by main.tex. Notation macros live in macros.tex.
% =============================================================================

% --- Encoding, fonts, language ----------------------------------------------
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage[british]{babel}
\usepackage{csquotes}
\usepackage{microtype}

% --- Layout -----------------------------------------------------------------
\usepackage[a4paper,margin=2.5cm]{geometry}
\usepackage{parskip}

% --- Mathematics ------------------------------------------------------------
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{amsthm}
\usepackage{mathtools}   % \DeclarePairedDelimiter, \coloneqq, \prescript, ...
\usepackage{bm}          % \bm{} bold maths symbols
\usepackage{mathrsfs}    % \mathscr{}
\usepackage{dsfont}      % \mathds{1} indicator
\usepackage{thmtools}    % theorem styling and lists
\usepackage{txfonts}

% --- Floats, tables, graphics -----------------------------------------------
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{siunitx}
\usepackage{subcaption}
\usepackage{xcolor}
\usepackage{tikz}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

% --- Lists ------------------------------------------------------------------
\usepackage{enumitem}
\setlist{nosep}

% --- Bibliography -----------------------------------------------------------
\usepackage[
  backend=biber,
  style=authoryear,
  maxcitenames=2,
  maxbibnames=10,
  giveninits=true,
  uniquename=false,
  uniquelist=false,
]{biblatex}

% --- Multi-file editing -----------------------------------------------------
% Each file in sections/ starts with \documentclass[../main.tex]{subfiles}
% and can be compiled on its own, inheriting this preamble.
\usepackage{subfiles}

% --- Cross-referencing (hyperref and cleveref must come last) ---------------
\usepackage[
  colorlinks=true,
  linkcolor=blue!60!black,
  citecolor=green!40!black,
  urlcolor=blue!60!black,
]{hyperref}
\usepackage[capitalise,noabbrev]{cleveref}

% --- Theorem environments ---------------------------------------------------
% Numbered within sections, sharing a single counter so that Theorem 2.1 is
% followed by Lemma 2.2, etc.
\declaretheorem[style=plain,numberwithin=section]{theorem}
\declaretheorem[style=plain,sibling=theorem]{lemma}
\declaretheorem[style=plain,sibling=theorem]{proposition}
\declaretheorem[style=plain,sibling=theorem]{corollary}
\declaretheorem[style=plain,sibling=theorem]{conjecture}
\declaretheorem[style=definition,sibling=theorem]{definition}
\declaretheorem[style=definition,sibling=theorem]{assumption}
\declaretheorem[style=definition,sibling=theorem]{example}
\declaretheorem[style=remark,sibling=theorem]{remark}

% --- Equation numbering -----------------------------------------------------
\numberwithin{equation}{section}

% --- Misc -------------------------------------------------------------------
\allowdisplaybreaks   % let long align environments break across pages
