\begin{definition}[Placeholder]
\label{def:placeholder}
Let $\vb{x} \in \R^{n}$ and let $p(\vb{x})$ be a density. Write
\begin{equation}
\label{eq:placeholder}
  \E[f(\vb{x})] \defeq \int f(\vb{x})\, p(\vb{x}) \,\dd\vb{x},
  \qquad
  \norm*{\vb{x}}_{2}^{2} = \inner{\vb{x}}{\vb{x}}.
\end{equation}
\end{definition}

\label{ass:placeholder}
The samples satisfy $\vb{x}_{1}, \dots, \vb{x}_{N} \iid p$.
\end{assumption}
